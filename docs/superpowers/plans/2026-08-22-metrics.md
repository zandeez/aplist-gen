# Metrics Recording (VictoriaMetrics) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a dedicated `aplist-metrics` job that polls own-radio and external neighbor-scan state every 15 minutes and pushes it to a new VictoriaMetrics service, as observational history for future data-driven channel planning — with zero changes to the existing `aplist-gen`/`aplist-plan` jobs.

**Architecture:** A new pure-function module (`metrics.py`) formats poll results into Prometheus exposition text; a new CLI subcommand (`aplist-gen metrics`) polls both own-radio and neighbor-scan data (with per-host failure tracking, since the existing `poll_all`/`poll_all_neighbors` don't expose that) and pushes the formatted text to VictoriaMetrics over HTTP. VictoriaMetrics itself deploys as a new Nomad service job with its own host volume, reachable only via Nomad-internal networking (no Traefik route).

**Tech Stack:** Python 3.12+, stdlib `urllib` for the metrics push (no new dependency), pytest, Nomad + podman for deployment, VictoriaMetrics (single-node image) for storage.

**Spec:** `docs/superpowers/specs/2026-08-22-metrics-design.md`

## Global Constraints

- New value types are `@dataclass(frozen=True)`, matching `core.py`'s existing style.
- No comments except where they explain a non-obvious WHY — never restate what code already says.
- Run tests with `uv run pytest` from the repo root.
- Zero changes to `run_generate`/`run_plan` or their existing behavior — this feature is purely additive (a new subcommand, new module, new job files).
- External-network metrics are aggregated **per channel**, never labeled by BSSID or ESSID — this cluster sees 100+ unique external devices in range, most transient, and a label with that many churning distinct values causes unbounded time-series growth in VictoriaMetrics.
- This is a single-node Nomad cluster (confirmed 2026-08-22) — `aplist-metrics` reaches VictoriaMetrics via `http://127.0.0.1:<port>`, not Nomad service-discovery DNS.

def parse_own_radio_tables(
    bssids: dict[str, str],
    essids: dict[str, str],
    channels: dict[str, str],
    host: str | None = None,
    users: dict[str, str] | None = None,
) -> list[APRecord]:
    users = users or {}
    records: list[APRecord] = []
    for suffix in sorted(bssids):
        radio = suffix.split(".", 1)[0]
        channel = channels.get(radio)
        if channel is None:
            continue
        num_users_raw = users.get(radio)
        num_users = int(num_users_raw) if num_users_raw is not None else None
        records.append(
            APRecord(
                bssid=bssids[suffix],
                essid=essids[suffix],
                channel=int(channel),
                host=host,
                num_users=num_users,
            )
        )
    return records
```

async def poll_ap(host: str, community: str, timeout: int = 3, retries: int = 1) -> list[APRecord]:
    """Poll one HP MSM AP's own radio config over SNMP.

    Targets the AP's own configured radios/VAPs directly (not the RF
    neighbor-scan table) since self-hearing over RF is not reliable --
    see memory note orb_ap_cluster_is_hp_msm.md for how this was derived.
    """
    engine = SnmpEngine()
    target = await UdpTransportTarget.create((host, 161), timeout=timeout, retries=retries)

    raw_bssids = await walk_bounded(engine, community, target, BSSID_OID)
    raw_essids = await walk_bounded(engine, community, target, ESSID_OID)
    raw_channels = await walk_bounded(engine, community, target, CHANNEL_OID)
    raw_users = await walk_bounded(engine, community, target, USERS_OID)

    bssids = {k: format_bssid(v) for k, v in raw_bssids.items()}
    essids = {k: v.prettyPrint() for k, v in raw_essids.items()}
    channels = {k: v.prettyPrint() for k, v in raw_channels.items()}
    users = {k: v.prettyPrint() for k, v in raw_users.items()}

    return parse_own_radio_tables(bssids, essids, channels, host=host, users=users)
```

```python
import urllib.request

from aplist_gen.core import APRecord, ScanEntry


def _band_of_channel(channel: int) -> str:
    return "24g" if channel <= 14 else "5g"


def format_own_radio_metrics(records: list[APRecord]) -> str:
    lines: list[str] = []
    for record in records:
        if record.channel is None:
            continue
        host = record.host or ""
        labels = f'host="{host}",bssid="{record.bssid}",essid="{record.essid}"'
        lines.append(f"aplist_own_channel{{{labels}}} {record.channel}")
        if record.num_users is not None:
            lines.append(f"aplist_connected_users{{{labels}}} {record.num_users}")
    return "".join(line + "\n" for line in lines)


def format_neighbor_metrics(sightings: list[ScanEntry]) -> str:
    counts: dict[int, int] = {}
    max_rssi: dict[int, int] = {}
    for sighting in sightings:
        counts[sighting.channel] = counts.get(sighting.channel, 0) + 1
        if sighting.rssi is not None:
            max_rssi[sighting.channel] = max(max_rssi.get(sighting.channel, sighting.rssi), sighting.rssi)

    lines: list[str] = []
    for channel in sorted(counts):
        band = _band_of_channel(channel)
        labels = f'channel="{channel}",band="{band}"'
        lines.append(f"aplist_external_network_count{{{labels}}} {counts[channel]}")
        if channel in max_rssi:
            lines.append(f"aplist_external_signal_dbm_max{{{labels}}} {max_rssi[channel]}")
    return "".join(line + "\n" for line in lines)


def format_poll_failures(hosts: list[str], failed_hosts: list[str]) -> str:
    failed = set(failed_hosts)
    lines = [f'aplist_poll_failure{{host="{host}"}} {1 if host in failed else 0}' for host in hosts]
    return "".join(line + "\n" for line in lines)


def push_metrics(url: str, text: str) -> None:
    request = urllib.request.Request(url, data=text.encode("utf-8"), method="POST")
    with urllib.request.urlopen(request, timeout=10) as response:
        response.read()
```

Add the new polling helpers and `run_metrics`, placed after `_gather_plan_data` and before `format_plan_report_html`:

```python
async def _poll_all_with_failures(hosts: list[str], community: str) -> tuple[list[APRecord], list[str]]:
    results = await asyncio.gather(
        *(poll_ap(host, community) for host in hosts), return_exceptions=True
    )
    records_by_bssid: dict[str, APRecord] = {}
    failed_hosts: list[str] = []
    for host, result in zip(hosts, results):
        if isinstance(result, Exception):
            print(f"warning: poll of {host} failed: {result}", file=sys.stderr)
            failed_hosts.append(host)
            continue
        for record in result:
            records_by_bssid[record.bssid] = record
    return list(records_by_bssid.values()), failed_hosts


async def _poll_all_neighbors_with_failures(
    hosts: list[str], community: str, own_essid: str
) -> tuple[list[ScanEntry], list[str]]:
    results = await asyncio.gather(
        *(poll_neighbors(host, community, own_essid) for host in hosts), return_exceptions=True
    )
    entries_by_bssid: dict[str, ScanEntry] = {}
    failed_hosts: list[str] = []
    for host, result in zip(hosts, results):
        if isinstance(result, Exception):
            print(f"warning: neighbor scan of {host} failed: {result}", file=sys.stderr)
            failed_hosts.append(host)
            continue
        for entry in result:
            entries_by_bssid[entry.bssid] = entry
    return list(entries_by_bssid.values()), failed_hosts


async def _gather_metrics_data(
    hosts: list[str], community: str, target_essid: str
) -> tuple[list[APRecord], list[ScanEntry], list[str]]:
    (own_records, own_failed), (external_sightings, external_failed) = await asyncio.gather(
        _poll_all_with_failures(hosts, community),
        _poll_all_neighbors_with_failures(hosts, community, target_essid),
    )
    failed_hosts = sorted(set(own_failed) | set(external_failed))
    return own_records, external_sightings, failed_hosts


def run_metrics(args: argparse.Namespace) -> int:
    config = _read_ap_config()
    if config is None:
        return 1
    hosts, community, target_essid = config
    metrics_url = os.environ.get("METRICS_URL")

    own_records, external_sightings, failed_hosts = asyncio.run(
        _gather_metrics_data(hosts, community, target_essid)
    )
    own_records = [r for r in own_records if r.essid == target_essid]

    text = (
        format_own_radio_metrics(own_records)
        + format_neighbor_metrics(external_sightings)
        + format_poll_failures(hosts, failed_hosts)
    )

    if args.debug_dump:
        print(text, end="")

    if not metrics_url:
        print("warning: METRICS_URL is not set, skipping push", file=sys.stderr)
        return 0

    try:
        push_metrics(metrics_url, text)
    except OSError as exc:
        print(f"warning: failed to push metrics to {metrics_url}: {exc}", file=sys.stderr)

    return 0
```
